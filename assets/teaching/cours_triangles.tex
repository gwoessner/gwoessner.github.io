\section{Constructions de triangles}

\begin{prop}{Méthodes de construction}{}
	Pour \textbf{construire un triangle}, il faut connaître certaines de ces dimensions:
		\begin{enumerate}
		\item On connaît la \textbf{longueur} de ses \textbf{trois} côtés. On a besoin de \trou{règle graduée} et \trou{compas}.
		\begin{center}
			\includegraphics[width=0.2\linewidth]{triangle0001}
		\end{center}
		\item On connaît les mesures de \textbf{deux angles} et \textbf{la longueur} du côté situé entre eux. On a besoin de \trou{règle graduée} et \trou{rapporteur}.
		\begin{center}
			\includegraphics[width=0.15\linewidth]{triangle0002}
		\end{center}
		\item On connaît les \textbf{longueurs de deux côtés} et la mesure de \textbf{l'angle compris entre ces deux côtés}. On a besoin de \trou{règle graduée} et \trou{compas} et \trou{rapporteur}.
		\begin{center}
			\includegraphics[width=0.15\linewidth]{triangle0003}
	\end{center}
\end{enumerate}
\end{prop}


\begin{prop}{Inégalité triangulaire}{}
	Dans un triangle, la longueur de chaque côté est inférieure à la somme des longueurs des deux autres côtés. Dans le triangle $ABC$ ci-dessous, on a :
	$AB < AC + BC$, $AC < AB + BC$ et $BC < AB + AC$. C'est ce qu'on appelle l'\trou{\textbf{inégalité triangulaire}}.
	\begin{center}
		\includegraphics[width=0.3\linewidth]{triangle1}
	\end{center}
\end{prop}

\begin{rem}{}{}
	Pour qu’un triangle soit constructible, il faut que la longueur du plus grand côté soit inférieure à la somme des deux autres.
\end{rem}

\begin{ex}{Applications}{}
Le triangle $ABC$ est-il constructible ?
\begin{enumerate}
	\item $AB=2,2$ cm, $AC=0,9$ cm et $BC=1,6$ cm.\\
	\trou{On sait que : $AB = 2,2$ cm, $BC + AC = 1,6 + 0,9 = 2,5$ cm, }\\ \trou{ soit : $2,5$ cm $> 2,2$ cm.
	Or, d’après l’inégalité triangulaire, le triangle $ABC$ existe.}
	\item $AB=2,2$ cm, $AC=0,9$ cm et $BC=0,8$ cm.\\
\trou{	On sait que : $AB = 2,2$ cm, $BC + AC = 0,8 + 0,9=1,7$ cm,}\\ \trou{soit : $1,7$ cm $< 2,2$ cm.
	Or, d’après l’inégalité triangulaire, le triangle $ABC$ n'existe pas.}
	\item $AB=2,2$ cm, $AC=0,9$ cm et $BC=1,3$ cm.\\
	\trou{On sait que : $AB = 2,2$ cm, $BC + AC = 1,3 + 0,9 = 1,3$ cm,} \\ \trou{soit : $2,2$ cm $= 2,2$ cm.
	Or, d’après l’inégalité triangulaire, le triangle $ABC$ est aplati.}
\end{enumerate}
\end{ex}


\begin{defin}{Nature d’un triangle (Rappel)}{}
\begin{enumerate}
	\item Triangle \trou{\textbf{rectangle}} en $A$: triangle possédant un angle droit en $A$. Dans un triangle rectangle, on appelle \textbf{hypoténuse} le côté opposé à l'angle droit.
	\item Triangle \trou{\textbf{isocèle}} en $A$: triangle ayant \textbf{deux} de ses côtés de même longueur.
	\item Triangle \trou{\textbf{équilatéral}}: triangle ayant \textbf{tous} ses côtés de même longueur.
\end{enumerate}

\end{defin}

\section{Droites remarquables d’un triangle}

\subsection{Hauteur}

\begin{defin}{}{}
	Dans un triangle, une \trou{\textbf{hauteur}} est une droite qui passe par un sommet et qui est \textbf{perpendiculaire} au côté opposé, qu'on appelle la \trou{\textbf{base}}. Attention ! La base n'est pas forcément le côté "en bas".
		\begin{center}
		\includegraphics[width=0.4\linewidth]{triangle9}
	\end{center}
\end{defin}

\begin{prop}{Aire d'un triangle}{}
L’aire d’un triangle est égale à la moitié du produit de la longueur d’un de ses côtés par la hauteur relative à ce côté: $\mathcal{A} = \displaystyle\frac{c \times h}{2}$, où
$\mathcal{A}$ est l’aire du triangle ;
$c$ est la longueur d’un des côtés du triangle ;
$h$ est la hauteur relative de ce côté.
\end{prop}

\begin{ex}{}{}
	Si $ABC$ est un triangle tel que $BC =4$ cm et $AH=2,5$ cm, alors l'aire $\mathcal{A}$ du triangle $ABC$ vaut:
	${A} = \displaystyle\frac{c \times h}{2}=$ \trou{$\displaystyle\frac{4 \times 2,5}{2}$}=\trou{$\displaystyle\frac{10}{2}$} = \trou{$5$} \hspace{0.1cm} $cm^2$
	\begin{center}
		\includegraphics[width=0.25\linewidth]{triangle10}
	\end{center}
\end{ex}

\begin{rem}{}{}
Il peut arriver que la hauteur soit à l'extérieur du triangle, mais la formule reste la même :
$$\mathcal{A} = \frac{c\times h}{2} = \frac{AB \times HC}{2}.$$
\begin{center}
		\includegraphics[width=0.25\linewidth, angle=0]{triangle11}
	\end{center}
\end{rem}

\subsection{Médiatrice}

\begin{defin}{}{}
	La \trou{\textbf{médiatrice}} d’un segment est la droite qui passe par son \textbf{milieu} et qui lui est \textbf{perpendiculaire}.
\end{defin}

\begin{prop}{Construction}{}
	\begin{enumerate}
		\item \textcolor{black}{On place le milieu $I$ du segment $[AB]$}.                                
   \item \textcolor{black}{On trace la perpendiculaire à $[AB]$ passant par $I$}.
\end{enumerate}
\begin{center}
	\includegraphics[width=0.25\linewidth]{triangle6}
\end{center}
\end{prop}

\begin{prop}{}{}
Tous les points situés sur la médiatrice de $[AB]$ sont à \trou{\textbf{égale}} distance de $A$ et de $B$. On dit qu’ils sont \textbf{équidistants} de A et de B.
	\begin{center}
		\includegraphics[width=0.5\linewidth]{triangle7}
	\end{center}
\end{prop}
