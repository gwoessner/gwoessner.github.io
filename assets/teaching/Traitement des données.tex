\section{Calculer des effectifs, des fréquences, une moyenne}

\begin{defin}{}{}
	Dans une série de données :\\
	- l'\textbf{effectif} d'une donnée est le nombre de fois où cette donnée apparaît.\\
	- l'\textbf{effectif total} est la somme de tous les effectifs.\\
	- la \textbf{fréquence} d'une donnée est le quotient de son effectif par l'effectif total:
	  \[ \text{fréquence d'une donnée} = \displaystyle\frac{\text{effectif de la donnée}}{\text{effectif total}}\]
\end{defin}


\begin{ex}{}{}
	Voici les réponses d'un groupe d'élèves à la question : "Quelle est votre couleur préférée ?" : \\
	\textcolor{blue}{bleu} - \textcolor{red}{rouge} - \textcolor{blue}{bleu} - \textcolor{olive}{vert} - \textcolor{purple}{violet} - \textcolor{blue}{bleu} - \textcolor{olive}{vert} -  \textcolor{red}{rouge} - \textcolor{olive}{vert} - \textcolor{olive}{vert} - \textcolor{purple}{violet} -\textcolor{purple}{violet} - \textcolor{pink}{rose} - \textcolor{olive}{vert} - \textcolor{orange}{orange} - \textcolor{blue}{bleu} - \textcolor{red}{rouge} - \textcolor{blue}{bleu} - \textcolor{orange}{orange} - \textcolor{olive}{vert}\\
	On peut regrouper cette série de données dans un tableau, appelé \textbf{tableau d'effectifs} :
	\begin{center}
				\includegraphics[width=0.85\linewidth]{don1}
	\end{center}
\end{ex}

\begin{rem}{}{}
	Les fréquences et les effectifs sont \trou{proportionnels}.\\
	Une fréquence peut être donnée sous forme de fraction, de nombre décimal ou de pourcentage; elle est comprise entre \trou{$0$} et \trou{$1$}.\\
	La somme de toutes les fréquences est égale à \trou{$1$}.
\end{rem}


\begin{defin}{}{}
	La \textbf{moyenne} d'une série de données est égale au quotient de la somme de ces données par l'effectif total:
	\[ \text{moyenne} = \displaystyle\frac{\text{somme des données}}{\text{effectif total}}\]
\end{defin}


\begin{rem}{}{}
	La moyenne n'est pas nécessairement égale à l'une des données.\\
	La moyenne est toujours comprise entre la plus petite et la plus grande valeur de la série. 
\end{rem}

\begin{ex}{}{}
	Voici les notes sur $20$ obtenues par Alice en mathématiques au premier trimestre: $11$ ; $12,5$ ; $14$ ; $9,5$ ; $13$ \\
	Pour calculer sa moyenne, on calcule la somme de ses notes, que l'on divise par le nombre de notes: $\displaystyle\frac{\trou{11+12,5+14+9,5+13}}{\trou{5}} = \displaystyle\frac{\trou{60}}{\trou{5}}=\trou{12}$\\
	La moyenne des notes d'Alice est égale à $12/20$.\\
	
	\vspace{0.05cm}
	Dans cet exemple: \includegraphics[width=0.50\linewidth]{don2}
\end{ex}

\section{Représenter graphiquement des données numériques et non numériques}

\begin{prop}{Diagramme en bâtons}{}
	Dans un \textbf{diagramme en bâtons}, la \textbf{hauteur} de chaque bâton est \textbf{proportionnelle} à l'\textbf{effectif} de la valeur qu'il représente.
\end{prop}

\begin{ex}{}{}
	Le professeur de mathématiques a relevé les notes de ses élèves au dernier contrôle:
	\begin{center}
				\includegraphics[width=0.85\linewidth]{don3}
	\end{center}
	Chaque note est représentée par un bâton, la hauteur du bâton correspond à \trou{l'effectif de la note} :
		\begin{center}
		\includegraphics[width=0.5\linewidth]{don4}
	\end{center}
\end{ex}


\begin{prop}{Histogramme}{}
	Quand les données sont nombreuses, on peut les regrouper en \textbf{classes} et les représenter par un \textbf{histogramme}.
\end{prop}

\begin{ex}{}{}
	Lors d'une visite médicale, on a mesuré la taille en centimètres des élèves d'une classe de cinquième. Comme les données sont nombreuses, elles ont été regroupées en classe d'\textbf{amplitude} $5$cm:
		\begin{center}
		\includegraphics[width=0.857\linewidth]{don5} 			\includegraphics[width=0.5\linewidth]{don6}
		\end{center}
\end{ex}




\begin{prop}{Diagramme circulaire}{}
	Dans un \textbf{diagramme circulaire}, les \textbf{mesures des angles sont proportionnelles aux effectifs} de chaque catégorie.
\end{prop}


\begin{ex}{}{}
	Voici la répartition des $100$ élèves de cinquième d'un collège selon leur seconde langue vivante:
			\begin{center}
		\includegraphics[width=0.75\linewidth]{don8}
	\end{center}
L'effectif total est $100$, il correspond à $360 ^{\circ}$ sur le diragramme circulaire.\\
Il suffit donc de multiplier chaque effectif par $3,6$ pour obtenir la mesure de l'angle correspondant.\\
On reporte les résultats obtenues dans le tableau, puis on construit le diagramme.
			\begin{center}
		\includegraphics[width=0.85\linewidth]{don9}
		\includegraphics[width=0.2\linewidth]{don10}
			\end{center}
\end{ex}

\begin{rem}{}{}
	On peut également constuire un diagramme semi-circulaire. La somme des mesures des angles est alors égale à $180 ^{\circ}$.\\
\end{rem}
