\section{Les nombres relatifs}	

\begin{defin}{}{}
	Un nombre \textbf{positif} est un nombre \trou{supérieur ou égal} à 0.\\
	Un nombre \textbf{négatif} est un nombre \trou{inférieur ou égal} à 0.
\end{defin}


\begin{ex}{}{}
	\begin{enumerate}
		\item Exemples de nombres \textbf{positifs}: $14$ ans, $25$ mètres, $56$ kg, $\pointille[0.6cm]$
		\item Exemples de nombres \textbf{négatifs}: $-287$: naissance d'Archimède: $287$ avant la naissance de J.C, $-3^{\circ}$ : température de $3^{\circ}$ en dessous de $0$.
		\item $0$ est le seul nombre à la fois positif et négatif.
	\end{enumerate}
\end{ex}

\begin{rem}{}{}
	Le signe $+$ n'est pas toujours noté: $+14$ s'écrit $14$ ou $+25$ s'écrit $25$.
\end{rem}

\begin{defin}{}{}
	On appelle \trou{\textbf{nombre relatif}}, tout nombre négatif ou positif.
\end{defin}

\section{La droite graduée}

\subsection{Représentation des nombres relatifs sur la droite graduée}

\begin{defin}{Droite graduée}{}
	Une \textbf{droite graduée} est une droite qui a:\\
- \trou{un sens (indiqué par une flèche)}\\
- \trou{une origine}\\
- \trou{une unité de longueur}, que l’on reporte régulièrement de part et d’autre de l’origine
(donnant des graduations).\\
~\\
Sur une droite graduée, chaque point est repéré par un nombre relatif que l’on appelle \trou{\textbf{abscisse du point}}.\\
La \textbf{distance à zéro} d’un nombre quelconque $a$ est la longueur du segment [OA], où A est le point
d’abscisse $a$ et O l’origine de la droite graduée.\\
La distance à zéro est toujours un nombre \trou{positif}.
\end{defin}

\begin{ex}{}{}
	\begin{center}
		\includegraphics[width=1\linewidth]{droite1}
	\end{center}
	On dit que l'abscisse de A est $3$ et on note A(3). La distance à zéro est égale à 3.\\
	Quelles sont les abscisses de B, C, D, E ? Quelle sont les distances à zéro ?
	\begin{center}
		\begin{tabular}{|c|c|c|c|c|}
			
			\hline
		Abscisses  & \trou{B} & \trou{C} & \trou{D} & \trou{E}  \\
			\hline
			Distances à zéro  & \trou{$4,5$} & \trou{$3$} & \trou{$5$} & \trou{$2,5$}  \\
			\hline
		\end{tabular}	
	\end{center}
	Placer les points F et G d'abscisses respectives $-4$ et $5,5$.
\end{ex}



\subsection{Opposé d’un nombre}

\begin{defin}{Droite graduée}{}
	Deux nombres sont \trou{\textbf{opposés}} lorsqu’ils ont la même distance à zéro et sont de signes contraires.
\end{defin}



\begin{ex}{}{}
\begin{center}
	\begin{tabular}{|c|c|c|c|c|c|}
		
		\hline
		L'opposé de $\pointille[0.6cm]$ & $3$ & $-2$ & $-6$ & $13$ & $0$ \\
		\hline
		est $\pointille[0.6cm]$ & \trou{$-3$} & \trou{$2$} & \trou{$6$} & \trou{$-13$} & \trou{$0$} \\
		\hline
	\end{tabular}	
\end{center}
\end{ex}

\begin{rem}{}{}
	Deux points dont les abscisses sont opposées sont situés à égale distance de l’origine.
\end{rem}

\section{Comparaison des nombres relatifs}

\begin{defin}{}{}
	\textbf{Ordre croissant}: \trou{du plus petit au plus grand}.\\
	\textbf{Ordre décroissant}: \trou{du plus grand au plus petit}.
\end{defin}

\begin{prop}{}{}
	Un nombre positif est toujours \trou{supérieur} à un nombre négatif.\\
	Si deux nombres sont positifs, le plus grand des deux est celui qui a la plus grande
	distance à 0.\\
	Si deux nombres sont négatifs, le plus grand des deux nombres est celui qui a la
	plus petite distance à 0.	
\end{prop}

\begin{ex}{}{}
	\begin{enumerate}
		\item Comparer $2,5$ et $5,5$: \trou{$2,5<5,5$}
		\item Comparer $1,8$ et $-3,2$: \trou{$-3,2<1,8$}
		\item Comparer $-1$ et $-2,5$: \trou{$-2,5<-1$}
		\item Ranger les nombres suivants dans l'ordre croissant: $ -4,03  ;  2,5  ;  -4,3  ;  -3,4  ;  2,9$ :\\
		 \trou{$-4,3 < -4,03 <-3,4 < 2,5< 2,9$}
	\end{enumerate}
\end{ex}

\section{Repère du plan}


\begin{defin}{}{}
	Un \trou{\textbf{repère}} du plan est formé par deux droites graduées de même origine. La droite
	horizontale est appelée \trou{\textbf{axe des abscisses}} et la droite verticale est appelée \trou{\textbf{axe des	ordonnées}}.\\
	Quand les deux droites sont perpendiculaires, on parle de \textbf{repère} \trou{\textbf{orthogonal}}.\\
	Dans un repère du plan, chaque point est repéré par deux nombres relatifs : ses \trou{\textbf{coordonnées}}. Le premier est l’\textbf{abscisse}, le second l’\textbf{ordonnée}. On note (abscisse;ordonnée).
\end{defin}

\begin{defin}{}{}
	\begin{center}
		\includegraphics[width=0.6\linewidth]{repere1}
	\end{center}
\end{defin}

\begin{ex}{}{}
	L’abscisse du point A est \trou{$3$}. L’ordonnée du point A est \trou{$2$}. Les coordonnées du point A se notent \trou{(2 ;3)}.\\
	Placer le point B de coordonnées (-2;1).
\end{ex}
