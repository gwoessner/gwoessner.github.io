\section{Additions et soustractions avec les nombres relatifs}

\begin{ex}{}{}
Aujourd'hui il fait 3°C. Pendant la nuit la température baissera de 5°C. Quelle température fera-t-il cette nuit ? Quel calcul permet de représenter la situation ?\\
\trou{\textbf{Il fera -2°C, qui est le résultat du calcul $3-5$.}}\\
Au matin la température va encore baisser de 3°C. Quelle température fera-t-il demain matin ? Quel calcul permet de représenter la situation ?\\
\trou{\textbf{Il fera -5°C, qui est le résultat du calcul $-2-3$.}}\\
L'après-midi la température va se réchauffer de 11°C. Quelle température fera-t-il demain matin ? Quel calcul permet de représenter la situation ?\\
\trou{\textbf{Il fera 6°C, qui est le résultat du calcul $-5+11$.}}\\
\end{ex}

\section{Calculs avec des parenthèses}

\begin{prop}{Règles des signes qui se suivent}{}
\begin{enumerate}
	\item $\textcolor{olive}{+} \textcolor{olive}{+} \rightarrow \textcolor{olive}{+}$ :  \textcolor{olive}{ajouter(+)} un \textcolor{olive}{gain(+)} donne un \textcolor{olive}{gain(+)}. Deux \textcolor{olive}{« + »} qui se suivent, deviennent un \textcolor{olive}{« + »}.
	
	\item $\textcolor{olive}{+} \textcolor{red}{-} \rightarrow \textcolor{red}{-} $ : \textcolor{olive}{ajouter(+)} une \textcolor{red}{perte(-)} donne une \textcolor{red}{perte(-)}.  Un \textcolor{olive}{« + »} suivit d'un \textcolor{red}{« - »} , devient un \textcolor{red}{« - »}.
	\item $\textcolor{red}{-} \textcolor{red}{-} \rightarrow \textcolor{olive}{+}$ :  \textcolor{red}{enlever(-)} une \textcolor{red}{perte(-)} est un \textcolor{olive}{gain(+)}. Deux \textcolor{red}{« - »} qui se suivent, deviennent un \textcolor{olive}{« + »}.
	\item $\textcolor{red}{-}  \textcolor{olive}{+} \rightarrow \textcolor{red}{-}$ : \textcolor{red}{enlever(-)} un \textcolor{olive}{gain(+)} est une \textcolor{red}{perte(-)}.  Un \textcolor{red}{« - »} suivi d'un \textcolor{olive}{« + »} , devient un \textcolor{red}{« - »}.
\end{enumerate}
\end{prop}


\begin{ex}{}{}
	\begin{enumerate}
		\item $3 + (+7)=$ \trou{$3+7=10$}
		\item $8+(-3)=$ \trou{$8-3=5$}
		\item $8-(-5)=$ \trou{$8+5=13$}
		\item $-2-(+4)=$ \trou{$-2-4=-6$}
	\end{enumerate}
\end{ex}

\begin{ex}{Attention aux parenthèses !}{}
	\begin{enumerate}
		\item $3-(1-5)=$ \trou{$3-(-4)=3+4=7$}
		\item $4+(-7+9)=$ \trou{$4+(+2)=4+2=6$}
		\item $(-3)-(-6+8)=$ \trou{$(-3)-(+2)=-3-2=-5$}
		\item $(-3+11)+(-7+2)=$ \trou{$(+8)+(-5)=8-5=3$}
	\end{enumerate}
\end{ex}
