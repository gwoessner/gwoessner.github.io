\section{Parallélogrammes}
	
	
	\begin{prop}{}{}
		Dans un parallélogramme, les côtés opposés sont \trou{parallèles}.\\
		Dans un parallélogramme, les côtés opposés ont \trou{la même longueur}.\\
				Dan un parallélogramme, les diagonales se \trou{coupent en leur milieu}.
		\begin{center}
			\includegraphics[width=0.42\linewidth]{paraa}
		\end{center}
	\end{prop}
	


	
	\begin{prop}{Conditions pour qu'un quadrilatère soit un parallélogramme :}{}
		Si un quadrilatère a ses côtés opposés de même longueur, alors c'est un parallélogramme.\\Si un quadrilatère a ses côtés opposés parallèles, alors c'est un parallélogramme.\\
		Si un quadilatère a des diagonales qui se coupent en leur milieu, alors c'est un parallélogramme.\\
		~\\
		Si un quadrilatère non croisé a \textbf{deux côtés parallèles et de même longueur}, alors ce quadrilatère est un parallélogramme.
	\end{prop}

\begin{ex}{}{}
		Quelle est la nature du quadrilatère $TRUC$ ci-dessous ? C'est un  \trou{parallélogramme}.
	\begin{center}
	\includegraphics[width=0.25\linewidth]{para5}
\end{center}
Justification :\\
~\\~\\~\\~\\
\end{ex}

\begin{prop}{Aire}{}
	L’\textbf{aire d’un parallélogramme} est égale au produit de la longueur d’un côté par la hauteur relative à ce côté.
		\begin{center}
		\includegraphics[width=0.45\linewidth]{aire1}
	\end{center}
\end{prop}

\begin{ex}{}{}
	Calculer l'aire du parallélogramme $ABCD$ sachant que $(AH)$ est la hauteur relative au côté $[BC]$, $BC = 6$ cm et $AH = 5$ cm.
			\begin{center}
		\includegraphics[width=0.25\linewidth]{aire2}
	\end{center}
L'aire de $ABCD$ est égale à : \trou{$BC \times AH = 6 \times 5 = 30$ $cm^2$}.
\end{ex}

	\section{Losanges}

	
	\begin{prop}{}{}
		Un losange est un parallélogramme, il possède donc toutes les propriétés des parallélogrammes.\\
		~\\
		Mais en plus :\\
		Dans un losange, les quatre cotés sont \trou{de même longueur}.\\
		Dans un losange, les diagonales sont \trou{perpendiculaires}.
				\begin{center}
			\includegraphics[width=0.25\linewidth]{los1}
		\end{center}
	\end{prop}
	
	
\begin{prop}{Conditions pour qu'un quadrilatère soit un parallélogramme :}{}	
Si un parallélogramme a des diagonales perpendiculaires, alors c'est un losange.\\ 
Si un parallélogramme a deux côtés consécutifs égaux, alors c'est un losange.
	\end{prop}

\begin{ex}{}{}
	Quelle est la nature du parallélogramme $ABCD$ ci-dessous ? C'est un  \trou{losange}.
	\begin{center}
		\includegraphics[width=0.25\linewidth]{los2}
	\end{center}
Justification :\\
~\\
\end{ex}

	\section{Rectangles}
	
	
	\begin{prop}{}{}
	Un rectangle est un parallélogramme, il possède donc toutes les propriétés du parallélogramme.\\
	~\\
	Mais en plus :\\
	Dans un rectangle, les diagonales sont \trou{de même longueur}.\\
	Dans un rectangle, les quatre angles sont \trou{droits}.
			\begin{center}                                              			\includegraphics[width=0.3\linewidth]{rect1}
	\end{center}
	\end{prop}

	
\begin{prop}{Conditions pour qu'un quadrilatère soit un parallélogramme :}{}	

 Si un parallélogramme possède un angle droit, alors c'est un rectangle.\\
Si un parallélogramme possède des diagonales de même longueur, alors c'est un rectangle.
	\end{prop}

~\\~\\
\end{ex}                                                           	
	

	
	\section{Carrés}
	
	\begin{prop}{}{}
		Un carré est à la fois un losange et un rectangle et est aussi un parallélogramme. Ainsi, le carré possède toutes les propriétés des losanges et des rectangles.\\~\\
		\begin{center}
			\includegraphics[width=0.25\linewidth]{carre1}
		\end{center}
	\end{prop}

\begin{prop}{Conditions pour qu'un quadrilatère soit un parallélogramme :}{}	
Toutes les combinaisons sont possibles. Il faut prouver que le quadrilatère est un losange et aussi un rectangle.
\end{prop}
